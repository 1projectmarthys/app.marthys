<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Inventarisit extends Model
{
    use HasFactory;
    protected $table = 'inventarisit1'; // Nama tabel di database
    protected $fillable = [
        'kode_barang',
        'nama_barang',
        'spesifikasi_barang',
        'tahun',
        'jumlah',
        'kondisi_barang',
        'keterangan',
    ];

    protected static function boot()
    {
        parent::boot();

        // Event untuk membuat kode barang secara otomatis
        static::creating(function ($model) {
            $latest = self::latest('id')->first(); // Ambil data terakhir berdasarkan ID
            $number = $latest ? intval(substr($latest->kode_barang, -4)) + 1 : 1; // Ambil nomor terakhir dan tambahkan 1
            $model->kode_barang = 'IT-INVNTR-01-' . str_pad($number, 4, '0', STR_PAD_LEFT); // Format kode barang
        });
    }
}
